    43→const MOIS_FR = [
    44→  'janvier', 'février', 'mars', 'avril', 'mai', 'juin',
    45→  'juillet', 'août', 'septembre', 'octobre', 'novembre', 'décembre',

    47→const JOURS_FR = ['Dimanche', 'Lundi', 'Mardi', 'Mercredi', 'Jeudi', 'Vendredi', 'Samedi'];
    48→
    49→function formatDateFr(iso: string) {
    50→  const d = new Date(iso);
    51→  return `${JOURS_FR[d.getDay()]} ${d.getDate()} ${MOIS_FR[d.getMonth()]} ${d.getFullYear()}`;
    52→}
    53→
    54→function getTimeUntil(target: Date) {
    55→  const diff = Math.max(0, target.getTime() - Date.now());
    56→  return {
    57→    jours: Math.floor(diff / 86400000),
    58→    heures: Math.floor((diff / 3600000) % 24),
    59→    minutes: Math.floor((diff / 60000) % 60),
    60→    secondes: Math.floor((diff / 1000) % 60),
    61→  };
    62→}

   207→export function NoirEtOrTemplate({ template: t, data, mode = 'preview', onClose, eventSlug }: SpecializedTemplateProps) {
   208→  const targetDate = useMemo(() => new Date(data.eventDate), [data.eventDate]);
   209→  const [countdown, setCountdown] = useState(getTimeUntil(targetDate));
   210→  const [rsvpName, setRsvpName] = useState('');
   211→  const [rsvpGuests, setRsvpGuests] = useState('1');
   212→  const [rsvpNote, setRsvpNote] = useState('');
   213→  const [rsvpStatus, setRsvpStatus] = useState<'idle' | 'loading' | 'success' | 'error'>('idle');
   214→  const [rsvpError, setRsvpError] = useState<string | null>(null);
   215→
   216→  // Gold color from template or fallback
   217→  const gold = t.goldColor ?? COLORS.gold;
   218→  const cream = COLORS.cream;
   219→
   220→  // Countdown — update every second
   221→  useEffect(() => {
   222→    const iv = setInterval(() => setCountdown(getTimeUntil(targetDate)), 1000);
   223→    return () => clearInterval(iv);
   224→  }, [targetDate]);
   225→
   226→  // Monogram
   227→  const monogram = data.partner1Name[0] + '&' + data.partner2Name[0];
   228→
   229→  // RSVP handler
   230→  const handleRsvp = useCallback(async (e: React.FormEvent) => {
   231→    e.preventDefault();
   232→    if (mode === 'preview') {
   233→      setRsvpStatus('loading');
   234→      setTimeout(() => setRsvpStatus('success'), 800);
   235→      return;
   236→    }
   237→    if (!eventSlug) {
   238→      setRsvpStatus('error');
   239→      setRsvpError("Slug de l'événement manquant");
   240→      return;
   241→    }
   242→    setRsvpStatus('loading');
   243→    setRsvpError(null);
   244→    try {
   245→      const res = await fetch(`/api/events/${eventSlug}/rsvp`, {
   246→        method: 'POST',
   247→        headers: { 'Content-Type': 'application/json' },
   248→        body: JSON.stringify({
   249→          name: rsvpName,
   250→          plusOnes: parseInt(rsvpGuests, 10) - 1,
   251→          note: rsvpNote,
   252→        }),
   253→      });
   254→      if (!res.ok) {
   255→        const body = await res.json().catch(() => ({}));
   256→        throw new Error(body.error ?? `Erreur ${res.status}`);
   257→      }
   258→      setRsvpStatus('success');
   259→    } catch (err) {
   260→      setRsvpStatus('error');
   261→      setRsvpError(err instanceof Error ? err.message : 'Une erreur est survenue');
   262→    }
   263→  }, [mode, eventSlug, rsvpName, rsvpGuests, rsvpNote]);
   264→
   265→  return (
   266→    <div className="min-h-screen" style={{ background: COLORS.black, fontFamily: 'Georgia, serif' }}>
   267→      {/* ─── Close Button (preview) ────────────────────────────────────── */}
   268→      {mode === 'preview' && onClose && (
   269→        <div className="fixed top-4 right-4 z-50">
   270→          <button
   271→            onClick={onClose}
   272→            className="flex items-center justify-center rounded-full p-2 backdrop-blur-sm transition-all hover:opacity-80"
   273→            style={{ background: 'rgba(201, 168, 76, 0.15)', border: `1px solid ${gold}30` }}
   274→            aria-label="Fermer"
   275→          >
   276→            <X className="h-5 w-5" style={{ color: cream }} />
   277→          </button>
   278→        </div>
   279→      )}
   280→
   281→      {/* ══════════════════════════════════════════════════════════════════
   282→          SECTION 1 — COVER
   283→          ══════════════════════════════════════════════════════════════════ */}
   284→      <section className="relative flex min-h-screen flex-col items-center justify-center overflow-hidden">
   285→        {/* Background layers */}
   286→        <div className="absolute inset-0" style={{ background: COLORS.black }} />
   287→        {/* Subtle gold radial gradient overlay */}
   288→        <div
   289→          className="absolute inset-0"
   290→          style={{
   291→            background: `radial-gradient(ellipse at center, rgba(201,168,76,0.08) 0%, transparent 70%)`,
   292→          }}
   293→        />
   294→        {/* Art Deco frame */}
   295→        <div className="absolute inset-4 sm:inset-8 pointer-events-none">
   296→          <ArtDecoFrame gold={gold} />
   297→        </div>
   298→
   299→        <div className="relative z-10 flex flex-col items-center justify-center px-6 text-center">
   300→          {/* Crown monogram */}
   301→          <motion.div initial={{ opacity: 0, scale: 0.5 }} animate={{ opacity: 1, scale: 1 }} transition={{ duration: 1 }}>
   302→            <CrownMonogram monogram={monogram} gold={gold} />
   303→          </motion.div>
   304→
   305→          {/* Names */}
   306→          <motion.div initial="hidden" animate="visible" variants={stagger} className="mt-6 flex flex-col items-center">
   307→            <motion.p
   308→              variants={fadeInUp}
   309→              className="text-4xl sm:text-5xl md:text-6xl lg:text-7xl italic"
   310→              style={{ color: cream, textShadow: `0 0 40px ${gold}20` }}
   311→            >
   312→              {data.partner1Name}

   316→              <span className="text-2xl sm:text-3xl italic" style={{ color: gold }}>&amp;</span>
   317→              <div className="h-px w-10 sm:w-16" style={{ background: `${gold}60` }} />
   318→            </motion.div>
   319→            <motion.p
   320→              variants={fadeInUp}

   322→              style={{ color: cream, textShadow: `0 0 40px ${gold}20` }}

   339→            style={{ color: COLORS.creamMuted }}
   340→          >
   341→            {formatDateFr(data.eventDate)}

   343→
   344→          {/* Scroll indicator */}
   345→          <motion.div

   363→        <motion.div

   417→        <div className="max-w-md w-full text-center">

   420→            initial="hidden"
