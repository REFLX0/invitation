    91→// (Date formatting now handled by `useT().formatDate()` — see src/lib/i18n/use-t.ts)
    92→
    93→function getTimeUntil(target: Date) {
    94→  const diff = Math.max(0, target.getTime() - Date.now());
    95→  return {
    96→    jours: Math.floor(diff / 86400000),
    97→    heures: Math.floor((diff / 3600000) % 24),
    98→    minutes: Math.floor((diff / 60000) % 60),
    99→    secondes: Math.floor((diff / 1000) % 60),
   100→  };
   101→}

   178→  const [rsvpError, setRsvpError] = useState<string | null>(null);
   179→

   190→      setTimeout(() => setRsvpStatus('success'), 800);
   191→      return;
   192→    }

   194→      setRsvpStatus('error');
   195→      setRsvpError(t('rsvp.missingSlug'));

   205→          name: rsvpName,
   206→          plusOnes: parseInt(rsvpGuests, 10) - 1, // form sends total headcount; store +1s
   207→          note: rsvpNote,
   208→        }),

   211→        const body = await res.json().catch(() => ({}));
   212→        throw new Error(body.error ?? `${res.status}`);

   214→      setRsvpStatus('success');

   216→      setRsvpStatus('error');
   217→      setRsvpError(err instanceof Error ? err.message : t('rsvp.error'));

   245→
