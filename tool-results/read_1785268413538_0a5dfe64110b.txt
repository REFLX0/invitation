    91→];

    97→function formatDateFrench(dateStr: string): string {
    98→  const date = new Date(dateStr + 'T00:00:00');
    99→  if (isNaN(date.getTime())) return dateStr;
   100→  const day = FRENCH_DAYS[date.getDay()];
   101→  const num = date.getDate();
   102→  const month = FRENCH_MONTHS[date.getMonth()];
   103→  const year = date.getFullYear();
   104→  return `${day} ${num} ${month} ${year}`;
   105→}

   122→    opacity: 0,

   135→
   136→export function WizardModal({ open, onOpenChange }: WizardModalProps) {
   137→  const [step, setStep] = useState(0);
   138→  const [direction, setDirection] = useState(0);
   139→  const [selectedTemplate, setSelectedTemplate] = useState<Template | null>(null);
   140→  const [formData, setFormData] = useState<FormData>({
   141→    partner1Name: '',
   142→    partner2Name: '',
   143→    eventDate: '',
   144→    venueName: '',
   145→    venueAddress: '',
   146→  });
   147→  const [isSubmitting, setIsSubmitting] = useState(false);
   148→  const [submitError, setSubmitError] = useState<string | null>(null);
   149→  const [createdSlug, setCreatedSlug] = useState<string | null>(null);
   150→
   151→  /* ---- Reset on close ---- */
   152→  const handleClose = useCallback(
   153→    (value: boolean) => {
   154→      if (!value) {
   155→        // Reset state after animation completes
   156→        setTimeout(() => {
   157→          setStep(0);
   158→          setDirection(0);
   159→          setSelectedTemplate(null);
   160→          setFormData({
   161→            partner1Name: '',
   162→            partner2Name: '',
   163→            eventDate: '',
   164→            venueName: '',
   165→            venueAddress: '',
   166→          });
   167→          setIsSubmitting(false);
   168→          setSubmitError(null);
   169→          setCreatedSlug(null);
   170→        }, 200);
   171→      }
   172→      onOpenChange(value);
   173→    },
   174→    [onOpenChange],
   175→  );
   176→
   177→  /* ---- Navigation ---- */
   178→  const goNext = () => {
   179→    if (step < STEP_COUNT - 1) {
   180→      setDirection(1);
   181→      setStep((s) => s + 1);
   182→    }
   183→  };
   184→
   185→  const goBack = () => {
   186→    if (step > 0) {
   187→      setDirection(-1);
   188→      setStep((s) => s - 1);
   189→    }
   190→  };
   191→
   192→  /* ---- Validation ---- */
   193→  const isStepValid = (): boolean => {
   194→    switch (step) {
   195→      case 0:
   196→        return selectedTemplate !== null;
   197→      case 1:
   198→        return (
   199→          formData.partner1Name.trim() !== '' &&
   200→          formData.partner2Name.trim() !== '' &&
   201→          formData.eventDate !== ''
   202→        );
   203→      case 2:
   204→        return true;
   205→      case 3:
   206→        return createdSlug !== null;
   207→      default:
   208→        return false;
   209→    }
   210→  };
   211→
   212→  /* ---- Submit ---- */
   213→  const handleSubmit = async () => {
   214→    setIsSubmitting(true);
   215→    setSubmitError(null);
   216→
   217→    try {
   218→      const isoDate = new Date(formData.eventDate + 'T12:00:00').toISOString();
   219→
   220→      const res = await fetch('/api/events', {
   221→        method: 'POST',
   222→        headers: { 'Content-Type': 'application/json' },
   223→        body: JSON.stringify({
   224→          ...formData,
   225→          eventDate: isoDate,
   226→          templateId: selectedTemplate!.id,
   227→          type: 'WEDDING',
   228→        }),
   229→      });
   230→
   231→      if (!res.ok) {
   232→        const body = await res.json().catch(() => ({}));
   233→        throw new Error(body.error ?? `Erreur ${res.status}`);
   234→      }
   235→
   236→      const event = await res.json();
   237→      setCreatedSlug(event.slug);
   238→
   239→      // Persist edit token in localStorage so the owner can later manage their
   240→      // invitation(s) from the dashboard. The token is returned ONCE by the API.
   241→      if (event.editToken && typeof window !== 'undefined') {
   242→        try {
   243→          const key = 'daawa:events';
   244→          const existing = JSON.parse(localStorage.getItem(key) ?? '[]') as Array<{
   245→            slug: string;
   246→            editToken: string;
   247→            partner1Name: string;
   248→            partner2Name: string;
   249→            createdAt: string;
   250→          }>;
   251→          existing.push({
   252→            slug: event.slug,
   253→            editToken: event.editToken,
   254→            partner1Name: formData.partner1Name,
   255→            partner2Name: formData.partner2Name,
   256→            createdAt: new Date().toISOString(),
   257→          });
   258→          localStorage.setItem(key, JSON.stringify(existing));
   259→        } catch {
   260→          // localStorage may be unavailable (private mode, etc.) — non-fatal
   261→        }
   262→      }
   263→    } catch (err) {
   264→      setSubmitError(
   265→        err instanceof Error ? err.message : 'Une erreur inattendue est survenue',
   266→      );
   267→    } finally {
   268→      setIsSubmitting(false);
   269→    }
   270→  };
   271→
   272→  /* ---- WhatsApp share ---- */
   273→  const shareWhatsApp = () => {
   274→    if (!createdSlug) return;
   275→    const url = `${window.location.origin}/${createdSlug}`;
   276→    const message = `Vous etes invite(e) au mariage de ${formData.partner1Name} & ${formData.partner2Name} !\n\n${url}`;
   277→    window.open(`https://wa.me/?text=${encodeURIComponent(message)}`, '_blank');
   278→  };
   279→
   280→  /* ---- Form change handler ---- */
   281→  const updateField = (field: keyof FormData, value: string) => {
   282→    setFormData((prev) => ({ ...prev, [field]: value }));
   283→  };
   284→
   285→  /* ====================================================================== */
   286→  /*  RENDER                                                                */
   287→  /* ====================================================================== */
   288→
   289→  return (
   290→    <Dialog open={open} onOpenChange={handleClose}>
   291→      <DialogContent
   292→        showCloseButton={step !== 3 || !createdSlug}
   293→        className="
   294→          fixed inset-0 top-0 left-0 z-50 flex items-center justify-center
   295→          max-w-full h-full w-full max-h-full
   296→          translate-x-0 translate-y-0
   297→          rounded-none border-0 p-0
   298→          bg-cream
   299→          data-[state=open]:zoom-in-[1] data-[state=closed]:zoom-out-[1]
   300→        "
   301→      >
   302→        {/* Accessibility: hidden titles for screen readers */}
   303→        <DialogTitle className="sr-only">
   304→          {STEP_TITLES[step]} — Daawa
   305→        </DialogTitle>
   306→        <DialogDescription className="sr-only">
   307→          {STEP_DESCRIPTIONS[step]}
   308→        </DialogDescription>
   309→
   310→        <div className="flex h-full w-full flex-col overflow-hidden">
   311→          {/* ---- Header ---- */}
   312→          <header className="shrink-0 border-b border-line bg-cream px-6 py-4">
   313→            {/* Logo row */}
   314→            <div className="flex items-center justify-between">
   315→              <div className="flex items-center gap-3">
   316→                <div className="flex h-8 w-8 items-center justify-center rounded-full bg-burgundy">
   317→                  <Sparkles className="h-4 w-4 text-gold" />
   318→                </div>
   319→                <span className="font-display text-sm font-semibold tracking-wider text-burgundy uppercase">
   320→                  Daawa
   321→                </span>
   322→              </div>
   323→              <span className="text-xs font-medium uppercase tracking-widest text-ink/40">
   324→                Etape {step + 1} / {STEP_COUNT}
   325→              </span>
   326→            </div>
   327→
   328→            {/* Step title */}
   329→            <div className="mt-3">
   330→              <h2 className="font-display text-xl font-semibold tracking-wide text-burgundy">
   331→                {STEP_TITLES[step]}
   332→              </h2>
   333→              <p className="mt-0.5 font-serif text-sm italic text-ink/50">
   334→                {STEP_DESCRIPTIONS[step]}
   335→              </p>
   336→            </div>
   337→
   338→            {/* Step indicators */}
   339→            <div className="mt-4 flex items-center gap-2">
   340→              {Array.from({ length: STEP_COUNT }, (_, i) => (
   341→                <div
   342→                  key={i}
   343→                  className={cn(
   344→                    'h-2 rounded-full transition-all duration-300',
   345→                    i <= step
   346→                      ? 'bg-burgundy w-8'
   347→                      : 'bg-line w-2',
   348→                  )}
   349→                />
   350→              ))}
   351→            </div>
   352→          </header>
   353→
   354→          {/* ---- Step Content (scrollable) ---- */}
   355→          <div className="relative flex-1 overflow-y-auto">
   356→            <AnimatePresence initial={false} custom={direction} mode="wait">
   357→              <motion.div
   358→                key={step}
   359→                custom={direction}
   360→                variants={slideVariants}
   361→                initial="enter"
   362→                animate="center"
   363→                exit="exit"
   364→                transition={slideTransition}
   365→                className="absolute inset-0 overflow-y-auto px-6 py-8"
   366→              >
   367→                {step === 0 && <StepChooseTemplate selected={selectedTemplate} onSelect={setSelectedTemplate} />}
   368→                {step === 1 && <StepYourDetails data={formData} onChange={updateField} />}
   369→                {step === 2 && selectedTemplate && <StepPreview template={selectedTemplate} data={formData} />}
   370→                {step === 3 && (
   371→                  <StepConfirmation
   372→                    data={formData}
   373→                    template={selectedTemplate}
   374→                    createdSlug={createdSlug}
   375→                    isSubmitting={isSubmitting}
   376→                    error={submitError}
   377→                    onSubmit={handleSubmit}
   378→                    onShareWhatsApp={shareWhatsApp}
   379→                  />
   380→                )}
   381→              </motion.div>
   382→            </AnimatePresence>
   383→          </div>
   384→
   385→          {/* ---- Footer / Navigation ---- */}
   386→          <footer className="shrink-0 border-t border-line bg-cream px-6 py-4">
   387→            <div className="flex items-center justify-between">
   388→              <Button
   389→                variant="ghost"
   390→                onClick={goBack}
   391→                disabled={step === 0 || isSubmitting}
   392→                className="gap-2 text-ink/60 hover:text-burgundy"
   393→              >
   394→                <ArrowLeft className="h-4 w-4" />
   395→                Retour
   396→              </Button>
   397→
   398→              {/* Conditional next / submit button */}
   399→              {step < STEP_COUNT - 1 ? (
   400→                <Button
   401→                  onClick={goNext}
   402→                  disabled={!isStepValid()}
   403→                  className="
   404→                    gap-2 rounded-full bg-burgundy px-6
   405→                    text-sm font-semibold uppercase tracking-wider text-white
   406→                    shadow-lg shadow-burgundy/20
   407→                    hover:bg-burgundy-light hover:shadow-xl
   408→                    disabled:opacity-40 disabled:shadow-none
   409→                  "
   410→                >
   411→                  Suivant
   412→                  <ArrowRight className="h-4 w-4" />
   413→                </Button>
   414→              ) : step === 3 && !createdSlug && !isSubmitting ? (
   415→                <Button
   416→                  onClick={handleSubmit}
   417→                  disabled={isSubmitting}
   418→                  className="
   419→                    gap-2 rounded-full bg-burgundy px-6
   420→                    text-sm font-semibold uppercase tracking-wider text-white
   421→                    shadow-lg shadow-burgundy/20
   422→                    hover:bg-burgundy-light hover:shadow-xl
   423→                    disabled:opacity-40 disabled:shadow-none
   424→                  "
   425→                >
   426→                  Creer mon invitation
   427→                  <Sparkles className="h-4 w-4" />
   428→                </Button>
   429→              ) : null}
   430→            </div>
   431→          </footer>
   432→        </div>
   433→      </DialogContent>
   434→    </Dialog>
   435→  );
   436→}

   438→/* ========================================================================= */
   439→/*  Step 1 — Choose Template                                                 */
   440→/* ========================================================================= */
   441→
   442→function StepChooseTemplate({
   443→  selected,
   444→  onSelect,
   445→}: {
   446→  selected: Template | null;
   447→  onSelect: (t: Template) => void;
   448→}) {
   449→  const [query, setQuery] = useState('');
   450→  const [activeCategory, setActiveCategory] = useState<Category | 'Tous'>('Tous');
   451→
   452→  const filtered = useMemo(() => {
   453→    const q = query.trim().toLowerCase();
   454→    return TEMPLATES.filter((t) => {
   455→      if (activeCategory !== 'Tous' && t.category !== activeCategory) return false;
   456→      if (q && !t.name.toLowerCase().includes(q) && !t.id.toLowerCase().includes(q)) return false;
   457→      return true;
   458→    });
   459→  }, [query, activeCategory]);
   460→
   461→  return (
   462→    <div className="flex flex-col gap-4">
   463→      {/* Search + count */}
   464→      <div className="flex flex-col sm:flex-row gap-3 sm:items-center sm:justify-between">
   465→        <div className="relative w-full sm:max-w-xs">
   466→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-ink/40 pointer-events-none" />
   467→          <input
   468→            type="text"
   469→            value={query}
   470→            onChange={(e) => setQuery(e.target.value)}
   471→            placeholder="Rechercher un modèle…"
   472→            className="w-full rounded-full border border-line bg-white py-2 pl-9 pr-4 text-xs text-ink placeholder:text-ink/40 focus:outline-none focus:border-burgundy/40"
   473→          />
   474→        </div>
   475→        <p className="text-[11px] uppercase tracking-wider text-ink/40">
   476→          {filtered.length} / {TEMPLATES.length} modèles
   477→        </p>
   478→      </div>
   479→
   480→      {/* Category pills */}
   481→      <div className="-mx-1 flex gap-1.5 overflow-x-auto scrollbar-hide px-1 py-0.5">
   482→        {ALL_CATEGORIES.map((cat) => (
   483→          <button
   484→            key={cat}
   485→            type="button"
   486→            onClick={() => setActiveCategory(cat)}
   487→            className={cn(
   488→              'whitespace-nowrap rounded-full px-3 py-1 text-[10px] font-medium uppercase tracking-wider transition-colors',
   489→              activeCategory === cat
   490→                ? 'bg-burgundy text-white'
   491→                : 'border border-line text-ink/50 hover:border-burgundy/40 hover:text-burgundy',
   492→            )}
   493→          >
   494→            {cat}
   495→          </button>
   496→        ))}
   497→      </div>
   498→
   499→      {/* Template grid (scrollable when many) */}
   500→      <div className="max-h-[55vh] overflow-y-auto scrollbar-hide -mx-1 px-1">
   501→        {filtered.length === 0 ? (
   502→          <div className="py-12 text-center">
   503→            <p className="text-sm text-ink/40">Aucun modèle ne correspond à votre recherche.</p>
   504→          </div>
   505→        ) : (
   506→          <div className="grid grid-cols-2 gap-4 sm:grid-cols-3">
   507→            {filtered.map((template) => {
   508→              const isSelected = selected?.id === template.id;
   509→              return (
   510→                <motion.button
   511→                  key={template.id}
   512→                  type="button"
   513→                  whileHover={{ y: -2 }}
   514→                  whileTap={{ scale: 0.98 }}
   515→                  onClick={() => onSelect(template)}
   516→                  className={cn(
   517→                    'group relative flex flex-col overflow-hidden rounded-xl border-2 transition-colors duration-200 text-left',
   518→                    isSelected
   519→                      ? 'border-burgundy shadow-lg shadow-burgundy/10'
   520→                      : 'border-transparent hover:border-line',
   521→                  )}
   522→                >
   523→                  {/* Card preview area */}
   524→                  <div
   525→                    className={cn(
   526→                      'relative flex h-36 flex-col items-center justify-center gap-1.5 bg-gradient-to-br p-3',
   527→                      template.gradient,
   528→                    )}
   529→                  >
   530→                    {/* Corner ornaments */}
   531→                    <div className="pointer-events-none absolute top-2 left-2 h-3 w-3 border-t border-l border-gold/30" />
   532→                    <div className="pointer-events-none absolute top-2 right-2 h-3 w-3 border-t border-r border-gold/30" />
   533→                    <div className="pointer-events-none absolute bottom-2 left-2 h-3 w-3 border-b border-l border-gold/30" />
   534→                    <div className="pointer-events-none absolute bottom-2 right-2 h-3 w-3 border-b border-r border-gold/30" />
   535→
   536→                    {/* Couple names preview */}
   537→                    <p className={cn('font-script text-base leading-tight', template.textColor)}>
   538→                      Ahmed &amp; Yasmine
   539→                    </p>
   540→                    <p className={cn('text-[9px] uppercase tracking-widest', template.subtextColor)}>
   541→                      15 Sept 2026
   542→                    </p>
   543→
   544→                    {/* Selected checkmark */}
   545→                    {isSelected && (
   546→                      <motion.div
   547→                        initial={{ scale: 0 }}
   548→                        animate={{ scale: 1 }}
   549→                        className="absolute top-1.5 right-1.5 flex h-5 w-5 items-center justify-center rounded-full bg-burgundy"
   550→                      >
   551→                        <Check className="h-3 w-3 text-white" />
   552→                      </motion.div>
   553→                    )}
   554→                  </div>
   555→
   556→                  {/* Card info */}
   557→                  <div className="flex items-center justify-between bg-white p-2.5">
   558→                    <div className="min-w-0">
   559→                      <p className="truncate text-xs font-semibold text-ink/80 leading-tight">
   560→                        {template.name}
   561→                      </p>
   562→                      <p className="mt-0.5 text-[9px] uppercase tracking-wider text-ink/40">
   563→                        {template.tier} · {template.category}
   564→                      </p>
   565→                    </div>
   566→                    <span className={cn(
   567→                      'shrink-0 text-xs font-bold',
   568→                      template.tier === 'Luxe' ? 'text-gold' : template.tier === 'Premium' ? 'text-burgundy' : 'text-ink/60',
   569→                    )}>
   570→                      {template.price} DT
   571→                    </span>
   572→                  </div>
   573→                </motion.button>
   574→              );
   575→            })}
   576→          </div>
   577→        )}
   578→      </div>
   579→    </div>
   580→  );
   581→}

   583→/* ========================================================================= */
   584→/*  Step 2 — Your Details                                                    */
   585→/* ========================================================================= */
   586→
   587→function StepYourDetails({
   588→  data,
   589→  onChange,
   590→}: {
   591→  data: FormData;
   592→  onChange: (field: keyof FormData, value: string) => void;
   593→}) {
   594→  const fields: { key: keyof FormData; label: string; type?: string; placeholder: string }[] = [
   595→    { key: 'partner1Name', label: 'Votre nom', placeholder: 'Ahmed' },
   596→    { key: 'partner2Name', label: "Nom de votre partenaire", placeholder: 'Yasmine' },
   597→    { key: 'eventDate', label: "Date de l'evenement", type: 'text', placeholder: '2026-09-15' },
