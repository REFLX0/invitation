   186→// ─── Helper: derive monogram from names ──────────────────────────────────
   187→function deriveMonogram(data: { partner1Name?: string; partner2Name?: string; monogram?: string }): string {
   188→  if (data.monogram) return data.monogram;
   189→  const p1 = (data.partner1Name ?? '').charAt(0).toUpperCase();
   190→  const p2 = (data.partner2Name ?? '').charAt(0).toUpperCase();
   191→  return p1 && p2 ? `${p1} ${p2}` : p1 || p2 || '';
   192→}

   194→// ─── Helper: format date as DD · MM · YYYY ────────────────────────────────
   195→function formatDateDots(dateStr: string): string {
   196→  const d = new Date(dateStr);
   197→  const dd = String(d.getDate()).padStart(2, '0');
   198→  const mm = String(d.getMonth() + 1).padStart(2, '0');
   199→  const yyyy = d.getFullYear();
   200→  return `${dd} \u00B7 ${mm} \u00B7 ${yyyy}`;
   201→}

   203→// ─── Helper: day of week (French) ─────────────────────────────────────────
   204→function dayOfWeek(dateStr: string): string {
   205→  const days = ['Dimanche','Lundi','Mardi','Mercredi','Jeudi','Vendredi','Samedi'];
   206→  return days[new Date(dateStr).getDay()] ?? '';
   207→}
   208→
   209→// ─── Helper: month name (French) ────────────────────────────────────────
   210→function monthName(dateStr: string): string {
   211→  const months = ['Janvier','Février','Mars','Avril','Mai','Juin','Juillet','Août','Septembre','Octobre','Novembre','Décembre'];
   212→  const d = new Date(dateStr);
   213→  return `${months[d.getMonth()]} ${d.getFullYear()}`;
   214→}
